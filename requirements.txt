# DiffusedDINOv1 — single environment for every experiment under ``experiments/``
# and every dataset script under ``datasets/``.
#
# Install:
#     python -m venv .venv
#     source .venv/bin/activate
#     pip install -r requirements.txt
#     pip install -e <path>/dinov3        # Meta AI DINOv3 reference impl
#
# `dinov3` is not on PyPI — either `pip install -e` it, or point `DINOV3_ROOT`
# at its source tree and the loaders add it to `sys.path` at runtime.
#
# Gated weights (DINOv3, RoBERTa-large) need a HuggingFace token before
# the first run:
#     export HF_TOKEN=<your_huggingface_token>
#
# Versions below are pinned to a verified-working environment
# (torch 2.8.0+cu128 / transformers 5.12.0); `pip freeze` that env for exact
# pins if you need to reproduce it byte-for-byte.
